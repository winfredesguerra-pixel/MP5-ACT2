let arr = [10, 25, 5, 18, 30]; console.log("Array:", arr); let searchValue = parseInt(prompt("Enter a value to search:")); let found = false; for (let i = 0; i < arr.length; i++) { if (arr[i] === searchValue) { console.log("Value found!"); console.log("Value:", arr[i]); console.log("Index:", i); found = true; break; } } if (!found) { console.log("Value does not exist in the array."); }
